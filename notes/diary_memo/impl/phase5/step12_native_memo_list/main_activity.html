class MainActivity : FlutterActivity() {
    private var assetPackDelivery: AssetPackDeliveryChannel? = null
 
    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        flutterEngine.platformViewsController.registry.registerViewFactory(
            NativeMemoListFactory.VIEW_TYPE,          // 名前
            NativeMemoListFactory(flutterEngine.dartExecutor.binaryMessenger),  // 作り手
        )
        assetPackDelivery = AssetPackDeliveryChannel(...)
    }
}

名前は Dart 側（native_memo_list.dart の _viewType）と同じ文字にしてある。作り手の中身は NativeMemoList.kt。
